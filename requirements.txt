# LiteLLM proxy — pinned for reproducible image builds.
# Chosen: 1.85.1 (latest upstream stable Docker/PyPI line as of 2026-05-21).
# Rationale: tracks upstream stable for latest fixes while keeping this image
# aligned with the official LiteLLM release cadence.
# Revisit on every Dependabot PR against this file.
litellm[proxy]==1.85.1

# OpenTelemetry — required when litellm_settings.callbacks: ["otel"] is set.
# LiteLLM imports from opentelemetry.sdk and the OTLP HTTP exporter; the
# `[proxy]` extra does not pull these in, so we add them explicitly.
# Pinned to a single 1.31.x stack so api / sdk / exporter stay aligned
# (mismatched versions across this trio are the usual breakage mode).
# 1.31.x is the lowest line whose importlib-metadata range is compatible
# with litellm's pin (>=8.5.0) — earlier 1.27/1.28 conflict.
opentelemetry-api==1.31.1
opentelemetry-sdk==1.31.1
opentelemetry-exporter-otlp-proto-http==1.31.1

# Prisma client — required when DATABASE_URL is set so LiteLLM can
# initialize Postgres for spend logs, virtual keys, user/team mgmt, and
# admin-UI auth. The `[proxy]` extra does NOT pull this. The Dockerfile
# additionally runs `prisma generate` against LiteLLM's schema.prisma to
# bake the engine binaries into the image.
prisma==0.11.0
